cmake_minimum_required(VERSION 3.13)
project(lab03_mpi LANGUAGES CXX)
find_package(MPI REQUIRED COMPONENTS CXX)
add_executable(matrix_mul_mpi src/main_mpi.cpp)
target_compile_features(matrix_mul_mpi PRIVATE cxx_std_11)
target_link_libraries(matrix_mul_mpi PRIVATE MPI::MPI_CXX)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(matrix_mul_mpi PRIVATE -Wall -Wextra -pedantic)
endif()
